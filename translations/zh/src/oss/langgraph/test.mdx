## 开始使用

由于许多 LangGraph 智能体都依赖 state，一个有用的模式是在每个测试中先创建图，然后在测试里使用新的 checkpointer 实例来编译它。

下面的示例展示了它如何与一个简单的线性图一起工作，这个图会依次经过 `node1` 和 `node2`。每个节点都会更新单个 state key `my_key`：

:::python
```python
import pytest

from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver

def create_graph() -> StateGraph:
    class MyState(TypedDict):
        my_key: str

    graph = StateGraph(MyState)
    graph.add_node("node1", lambda state: {"my_key": "hello from node1"})
    graph.add_node("node2", lambda state: {"my_key": "hello from node2"})
    graph.add_edge(START, "node1")
    graph.add_edge("node1", "node2")
    graph.add_edge("node2", END)
    return graph

def test_basic_agent_execution() -> None:
    checkpointer = MemorySaver()
    graph = create_graph()
    compiled_graph = graph.compile(checkpointer=checkpointer)
    result = compiled_graph.invoke(
        {"my_key": "initial_value"},
        config={"configurable": {"thread_id": "1"}}
    )
    assert result["my_key"] == "hello from node2"
```
:::

:::js
```ts
import { test, expect } from 'vitest';
import {
  StateGraph,
  StateSchema,
  START,
  END,
  MemorySaver,
} from '@langchain/langgraph';
import * as z from "zod";

const State = new StateSchema({
  my_key: z.string(),
});

const createGraph = () => {
  return new StateGraph(State)
    .addNode('node1', (state) => ({ my_key: 'hello from node1' }))
    .addNode('node2', (state) => ({ my_key: 'hello from node2' }))
    .addEdge(START, 'node1')
    .addEdge('node1', 'node2')
    .addEdge('node2', END);
};

test('basic agent execution', async () => {
  const uncompiledGraph = createGraph();
  const checkpointer = new MemorySaver();
  const compiledGraph = uncompiledGraph.compile({ checkpointer });
  const result = await compiledGraph.invoke(
    { my_key: 'initial_value' },
    { configurable: { thread_id: '1' } }
  );
  expect(result.my_key).toBe('hello from node2');
});
```
:::

## 测试单个节点和边

已编译的 LangGraph 智能体会把每个单独节点暴露为 `graph.nodes`。你可以利用这一点测试智能体中的单个节点。注意，这样会绕过编译图时传入的任何 checkpointer：

:::python
```python
import pytest

from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver

def create_graph() -> StateGraph:
    class MyState(TypedDict):
        my_key: str

    graph = StateGraph(MyState)
    graph.add_node("node1", lambda state: {"my_key": "hello from node1"})
    graph.add_node("node2", lambda state: {"my_key": "hello from node2"})
    graph.add_edge(START, "node1")
    graph.add_edge("node1", "node2")
    graph.add_edge("node2", END)
    return graph

def test_individual_node_execution() -> None:
    # 这个示例中会被忽略
    checkpointer = MemorySaver()
    graph = create_graph()
    compiled_graph = graph.compile(checkpointer=checkpointer)
    # 只调用 node 1
    result = compiled_graph.nodes["node1"].invoke(
        {"my_key": "initial_value"},
    )
    assert result["my_key"] == "hello from node1"
```
:::

:::js
```ts
import { test, expect } from 'vitest';
import {
  StateGraph,
  START,
  END,
  MemorySaver,
  StateSchema,
} from '@langchain/langgraph';
import * as z from "zod";

const State = new StateSchema({
  my_key: z.string(),
});

const createGraph = () => {
  return new StateGraph(State)
    .addNode('node1', (state) => ({ my_key: 'hello from node1' }))
    .addNode('node2', (state) => ({ my_key: 'hello from node2' }))
    .addEdge(START, 'node1')
    .addEdge('node1', 'node2')
    .addEdge('node2', END);
};

test('individual node execution', async () => {
  const uncompiledGraph = createGraph();
  // 这个示例中会被忽略
  const checkpointer = new MemorySaver();
  const compiledGraph = uncompiledGraph.compile({ checkpointer });
  // 只调用 node 1
  const result = await compiledGraph.nodes['node1'].invoke(
    { my_key: 'initial_value' },
  );
  expect(result.my_key).toBe('hello from node1');
});
```
:::

## 部分执行

对于由更大图组成的智能体，你可能希望测试智能体中某条部分执行路径，而不是端到端地测试整个流程。在某些情况下，把这些部分重构成子图可能更符合语义，这样你就可以像正常情况一样单独调用它们。

不过，如果你不想改动智能体图的整体结构，就可以使用 LangGraph 的持久化机制，模拟智能体在目标部分开始之前刚好暂停，并在目标部分结束时再次暂停。步骤如下：

:::python
1. 使用 checkpointer 编译智能体（用于测试时，内存 checkpointer @[`InMemorySaver`] 就足够了）。
2. 调用智能体的 [`update_state`](/oss/langgraph/use-time-travel) 方法，并把 [`as_node`](/oss/langgraph/use-time-travel#from-a-specific-node) 参数设置为你想要测试的节点之前那个节点的名称。
3. 使用同一个 `thread_id` 调用智能体，并把 `interrupt_after` 参数设置为你想要停止的节点名称。
:::
:::js
1. 使用 checkpointer 编译智能体（用于测试时，内存 checkpointer [`MemorySaver`](https://reference.langchain.com/javascript/classes/_langchain_langgraph-checkpoint.MemorySaver.html) 就足够了）。
2. 调用智能体的 [`update_state`](/oss/langgraph/use-time-travel) 方法，并把 [`asNode`](/oss/langgraph/use-time-travel#from-a-specific-node) 参数设置为你想要测试的节点之前那个节点的名称。
3. 使用同一个 `thread_id` 调用智能体，并把 `interruptBefore` 参数设置为你想要停止的节点名称。
:::

下面是一个只执行线性图中第二和第三个节点的示例：

:::python
```python
import pytest

from typing_extensions import TypedDict
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver

def create_graph() -> StateGraph:
    class MyState(TypedDict):
        my_key: str

    graph = StateGraph(MyState)
    graph.add_node("node1", lambda state: {"my_key": "hello from node1"})
    graph.add_node("node2", lambda state: {"my_key": "hello from node2"})
    graph.add_node("node3", lambda state: {"my_key": "hello from node3"})
    graph.add_node("node4", lambda state: {"my_key": "hello from node4"})
    graph.add_edge(START, "node1")
    graph.add_edge("node1", "node2")
    graph.add_edge("node2", "node3")
    graph.add_edge("node3", "node4")
    graph.add_edge("node4", END)
    return graph

def test_partial_execution_from_node2_to_node3() -> None:
    checkpointer = MemorySaver()
    graph = create_graph()
    compiled_graph = graph.compile(checkpointer=checkpointer)
    compiled_graph.update_state(
        config={
          "configurable": {
            "thread_id": "1"
          }
        },
        # 传入 node 2 的 state - 模拟位于
        # node 1 结束时
        values={"my_key": "initial_value"},
        # 把已保存的 state 更新为仿佛它来自 node 1
        # 执行会在 node 2 恢复
        as_node="node1",
    )
    result = compiled_graph.invoke(
        # 通过传入 None 继续执行
        None,
        config={"configurable": {"thread_id": "1"}},
        # 在 node 3 之后停止，这样 node 4 就不会运行
        interrupt_after="node3",
    )
    assert result["my_key"] == "hello from node3"
```
:::

:::js
```ts
import { test, expect } from 'vitest';
import {
  StateGraph,
  StateSchema,
  START,
  END,
  MemorySaver,
} from '@langchain/langgraph';
import * as z from "zod";

const State = new StateSchema({
  my_key: z.string(),
});

const createGraph = () => {
  return new StateGraph(State)
    .addNode('node1', (state) => ({ my_key: 'hello from node1' }))
    .addNode('node2', (state) => ({ my_key: 'hello from node2' }))
    .addNode('node3', (state) => ({ my_key: 'hello from node3' }))
    .addNode('node4', (state) => ({ my_key: 'hello from node4' }))
    .addEdge(START, 'node1')
    .addEdge('node1', 'node2')
    .addEdge('node2', 'node3')
    .addEdge('node3', 'node4')
    .addEdge('node4', END);
};

test('partial execution from node2 to node3', async () => {
  const uncompiledGraph = createGraph();
  const checkpointer = new MemorySaver();
  const compiledGraph = uncompiledGraph.compile({ checkpointer });
  await compiledGraph.updateState(
    { configurable: { thread_id: '1' } },
    // 传入 node 2 的 state - 模拟位于
    // node 1 结束时的状态
    { my_key: 'initial_value' },
    // 把已保存的 state 更新为仿佛它来自 node 1
    // 执行会在 node 2 恢复
    'node1',
  );
  const result = await compiledGraph.invoke(
    // 通过传入 None 继续执行
    null,
    {
      configurable: { thread_id: '1' },
      // 在 node 3 之后停止，这样 node 4 就不会运行
      interruptAfter: ['node3']
    },
  );
  expect(result.my_key).toBe('hello from node3');
});
```
